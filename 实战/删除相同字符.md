要求：删除字符串s1中所有s2中的字符

```c++
void squeeze (char s1[] , const char s2[]) {
    int i , j , k  ;
    for (j = 0 ; s2[j] != '\0' ; j ++) {
        for (i = k = 0 ; s1[i] != '\0' ; i ++) {
            if (s1[i] != s2[j] ) {
                s1[k++] = s1[i] ;
            }
        }
        s1[k] = '\0' ;
    }
}
```

简化版：删除字符串s中所有的特定字符

```c++
void squeeze (char s[] , char ch) {
	int i , j;
	for (i = j = 0 ; s[i] != '\0' ; i ++) {
		if (s[i] != ch) {
			s[j ++] = s[i];  //相当于从后往前覆盖
		}
	}
	s[j] = '\0';
}
```
